	--DBMS_OUTPUT.PUT_LINE('1');

	USP_EXCEPTION(''

				, v_UID
				);

	-- 회원 >> 회원관리 >> 대량업로드
	IF v_JOB_TYPE = 'USER_ExUpload' THEN
		INSERT INTO M_SCHEDULE_LOG (
				MSG
				,EXEDATE
		)VALUES('회원 대량업로드 시작_' || SYSDATE ||'_v_UID_'|| v_UID ||'_v_KEY_'|| v_KEY ||'_v_SITE_'|| v_SITE 
				,SYSDATE);        

		USP_USER_EXUPLOAD(v_UID,v_KEY);

	-- 회원 >> 회원관리 >> 일괄탈퇴
	ELSIF v_JOB_TYPE = 'WITHDRAW_USER_ExUpload' THEN
		INSERT INTO M_SCHEDULE_LOG (
				 MSG
				,EXEDATE
		)VALUES('회원 일괄탈퇴 시작_' || SYSDATE ||'_v_UID_'|| v_UID ||'_v_KEY_'|| v_KEY ||'_v_SITE_'|| v_SITE 
				,SYSDATE);    

		USP_WITHDRAW_USER_EXUPLOAD(v_UID,v_KEY);

	-- 회원 >> 회원관리 >> 회원정보 일괄수정
	ELSIF v_JOB_TYPE='USER_UPDATE_ExUpload' THEN
	     INSERT INTO M_SCHEDULE_LOG (MSG,EXEDATE)

	ELSIF v_JOB_TYPE='CORP_MANAGER_ExUpload' THEN

	     INSERT INTO M_SCHEDULE_LOG (MSG,EXEDATE)
	     VALUES('개인계약 승계처리 대량업로드 시작_' || SYSDATE ||'_v_UID_'|| v_UID ||'_v_KEY_'|| v_KEY ||'_v_SITE_'|| v_SITE , SYSDATE);    

	     USP_USER_SIGN_EXUPLOAD(v_UID,v_KEY);

	-- 계약 >> 법인계약 >> 승계처리 대량업로드
	ELSIF v_JOB_TYPE='CORP_SIGN_ExUpload' THEN
	     INSERT INTO M_SCHEDULE_LOG (MSG,EXEDATE)
	     VALUES('법인 승계처리 대량업로드 시작_' || SYSDATE ||'_v_UID_'|| v_UID ||'_v_KEY_'|| v_KEY ||'_v_SITE_'|| v_SITE , SYSDATE);    

	     USP_CORP_SIGN_EXUPLOAD(v_UID,v_KEY);

	-- 계약 >> 법인계약 >> 계약회원 대량업로드
	ELSIF v_JOB_TYPE='CORP_USER_SIGN_ExUpload' THEN
	     INSERT INTO M_SCHEDULE_LOG (MSG,EXEDATE)
	     VALUES('법인 계약회원 대량업로드 시작_' || SYSDATE ||'_v_UID_'|| v_UID ||'_v_KEY_'|| v_KEY ||'_v_SITE_'|| v_SITE , SYSDATE);    

	     USP_CORP_USER_SIGN_EXUPLOAD(v_UID,v_KEY);
	END IF;

	COMMIT;

	EXCEPTION
				WHEN OTHERS THEN
					--DECLARE  v_KEY NUMBER(10,0);
					USP_EXCEPTION( '', DBMS_UTILITY.FORMAT_ERROR_STACK , 'ExUpload', 'EXCEL_UPLOAD_MON', v_UID);
					ROLLBACK;

				BEGIN
					--utils.resetTrancount;

					UPDATE M_EXUP_TEMP
					SET  SF_FLAG = '실패'
						,ERROR_MSG = DBMS_UTILITY.FORMAT_ERROR_STACK
					WHERE M_EXUP_NO = v_KEY;

					COMMIT;

				END;
