		 SITE
		,USEQ
		,CORP
		,GBN
		,RT
		,DV
		,LOGIN_CNT
		,REGTIME
	)

		 L.SITE
		,L.USEQ
		,L.CORP
		,TRIM(L.GBN) AS GBN
		,L.RT
		,L.DV
		,COUNT(L.USEQ) AS LOGIN_CNT
		,TRUNC(L.REGTIME) AS REGTIME
	FROM A_LOGIN_LOG L
	WHERE 1 = 1

		AND L.SITE <> 'MS'
		AND L.SITE IN (SELECT S.SITE FROM A_SITE S)
		AND TRIM(L.GBN) LIKE '%로그인%'

		--AND L.REGTIME >= '2024-01-01'
		-- AND L.REGTIME >= TO_CHAR(TRUNC(SYSDATE - 2),'YYYY-MM-DD')
		AND L.REGTIME >= TRUNC(SYSDATE - 2)
		AND L.REGTIME < TRUNC(SYSDATE)
	GROUP BY
		L.SITE
		,L.USEQ
		,L.CORP
		,TRIM(L.GBN)
		,L.RT
		,L.DV
		,TRUNC(L.REGTIME)
	;

	/**** 현황 >> 서비스현황 >> 구독현황 *****/
	/*********************************************************************************************/

	--	,I.KEY1 AS RESNO

	--FROM A_FOLDER@DBL_CEO F

	MERGE INTO M_FOLDER_ITEM F
	USING (
			SELECT
				 F.FOLDERSEQ
				,I.FOLDERITEMSEQ
				,F.USEQ
				,F.SITE
				,F.TITLE
				,DECODE(F.TITLE,'_좋아요','P','L') AS GBN
				,I.MENUCODE
				,(SELECT M.MENUNAME FROM CEOMENU@DBL_CEO M WHERE M.MENUCODE = I.MENUCODE AND ROWNUM <= 1) AS MENUNAME
				,(SELECT G.GMENU FROM CEO_GNAV_NEW@DBL_CEO G WHERE G.MENUCODE = I.MENUCODE AND G.USEYN = 'Y' AND G.MENUCODE IS NOT NULL AND ROWNUM <= 1) AS GMENU
				,I.KEY1 AS RESNO
				,P.NAME AS RES_NAME
				,P.JIKGUBNM AS RES_JIKGBN
				,P.COMPANY AS RES_COMP
				--,F.REGTIME
				,I.REGTIME
				,NVL(I.CHGTIME,I.REGTIME) AS CHGTIME
			FROM A_FOLDER@DBL_CEO F
				INNER JOIN A_FOLDER_ITEM@DBL_CEO I 	ON I.FOLDERSEQ = F.FOLDERSEQ
				LEFT OUTER JOIN CEOPERSON@DBL_CEO P ON P.RESNO = I.KEY1
			WHERE 1 = 1
				AND F.SITE = 'CEO'
				AND F.USEYN = 'Y'
				AND F.OPENYN = 'N'
				AND F.CORP IS NULL
				AND F.TITLE IN ('_좋아요','_강사구독')
				AND F.USEQ IN (SELECT U.USEQ FROM A_USER U)
				AND I.USEYN = 'Y'
				-- AND NVL(I.CHGTIME,I.REGTIME) >= TO_CHAR(TRUNC(SYSDATE - 2),'YYYY-MM-DD')
				AND NVL(I.CHGTIME,I.REGTIME) >= TRUNC(SYSDATE - 2)
				AND NVL(I.CHGTIME,I.REGTIME) < TRUNC(SYSDATE)
			) C ON (F.FOLDERITEMSEQ = C.FOLDERITEMSEQ)
	WHEN MATCHED THEN
		UPDATE SET
			 F.FOLDERSEQ = C.FOLDERSEQ
			,F.USEQ = C.USEQ
			,F.SITE = C.SITE
			,F.TITLE = C.TITLE
			,F.GBN = C.GBN

			,F.MENUNAME = C.MENUNAME

			,F.RESNO = C.RESNO
			,F.RES_NAME = C.RES_NAME
			,F.RES_JIKGBN = C.RES_JIKGBN
			,F.RES_COMP = C.RES_COMP
			,F.REGTIME = C.REGTIME
			,F.CHGTIME = C.CHGTIME
	WHEN NOT MATCHED THEN
		INSERT (
			 F.FOLDERSEQ
			,F.FOLDERITEMSEQ
			,F.USEQ
			,F.SITE
			,F.TITLE
			,F.GBN
			,F.MENUCODE
			,F.MENUNAME
			,F.GMENU
			,F.RESNO
			,F.RES_NAME
			,F.RES_JIKGBN
			,F.RES_COMP
			,F.REGTIME
			,F.CHGTIME
		) VALUES (
			 C.FOLDERSEQ
			,C.FOLDERITEMSEQ
			,C.USEQ
			,C.SITE
			,C.TITLE
			,C.GBN
			,C.MENUCODE
			,C.MENUNAME
			,C.GMENU
			,C.RESNO
			,C.RES_NAME
			,C.RES_JIKGBN
			,C.RES_COMP
			,C.REGTIME
			,C.CHGTIME
		);

	/**** 집계 테이블 사용 쿼리 위치  *****/

			SELECT
				 P.REGSEQ
				,TRIM(P.USERID) AS USERID
				,U.USEQ
				,U.SITE

				,P.MENUCODE
				,P.CNT
				,P.GBN
				,P.USEYN
				,P.AMT
				,P.STATUS
				,P.PRICE
				,P.ETC
				,P.SEQ
				,P.CCNT
				,P.PCNT
				,P.TCNT
				,P.PAYMODE
				,P.PAYNAME
				,P.DEPOYN
				,P.ORDERNO
				--,P.REGDATE
				,TO_DATE(P.REGDATE,'YYYYMMDDHH24MISS') AS REGDATE
				,P.REGUSERID
				,DECODE(TRIM(P.REGUSERID)
						,TRIM(P.USERID), U.USEQ
						,(SELECT U.USEQ FROM A_USER U
							WHERE U.LOGINID = TRIM(P.REGUSERID)
								AND U.SITE = 'CEO'
								AND ROWNUM <= 1)
						) AS REGUSEQ
				--,P.CHGDATE
				,NVL(TO_DATE(P.CHGDATE,'YYYYMMDDHH24MISS')
					,TO_DATE(P.REGDATE,'YYYYMMDDHH24MISS')
					) AS CHGDATE
				,P.CHGUSERID
				,DECODE(TRIM(P.CHGUSERID)
						,TRIM(P.USERID), U.USEQ
						,NVL(
							(SELECT U.USEQ FROM A_USER U
							WHERE U.LOGINID = TRIM(P.CHGUSERID)
								AND U.SITE = 'CEO'
								AND ROWNUM <= 1)
							,(SELECT U.USEQ FROM A_USER U
							WHERE U.LOGINID = TRIM(P.REGUSERID)
								AND U.SITE = 'CEO'
								AND ROWNUM <= 1))
						) AS CHGUSEQ
			FROM CEO_PAYLOG@DBL_CEO P
				INNER JOIN A_USER U ON U.LOGINID = TRIM(P.USERID)
			WHERE 1 = 1
				AND MENUCODE = '0605'
				AND U.SITE = 'CEO'
				AND NVL(TRIM(P.USERID),'X') IN (SELECT U.LOGINID FROM A_USER U WHERE U.SITE = 'CEO')

				-- AND TO_DATE(NVL(P.CHGDATE,P.REGDATE),'YYYYMMDDHH24MISS') >= TO_CHAR(TRUNC(SYSDATE - 2),'YYYY-MM-DD')
				AND NVL(P.CHGDATE,P.REGDATE) >= TO_CHAR(TRUNC(SYSDATE - 2),'YYYYMMDDHH24MISS')
				-- AND NVL(TO_DATE(P.CHGDATE,'YYYYMMDDHH24MISS'),TO_DATE(P.REGDATE,'YYYYMMDDHH24MISS')) < TRUNC(SYSDATE)
				AND NVL(P.CHGDATE,P.REGDATE) < TO_CHAR(TRUNC(SYSDATE),'YYYYMMDDHH24MISS')
			) C ON (P.REGSEQ = C.REGSEQ)
	WHEN MATCHED THEN

			 P.USERID 		= C.USERID
			,P.USEQ 		= C.USEQ
			,P.SITE 		= C.SITE
			,P.MENUCODE 	= C.MENUCODE
			,P.CNT 			= C.CNT
			,P.GBN 			= C.GBN
			,P.USEYN 		= C.USEYN
			,P.AMT 			= C.AMT

			,P.ETC 			= C.ETC

			,P.PCNT 		= C.PCNT
			,P.TCNT 		= C.TCNT
			,P.PAYMODE 		= C.PAYMODE
			,P.PAYNAME 		= C.PAYNAME
			,P.DEPOYN 		= C.DEPOYN
			,P.ORDERNO 		= C.ORDERNO
			,P.REGDATE 		= C.REGDATE
			,P.REGUSERID 	= C.REGUSERID
			,P.REGUSEQ 		= C.REGUSEQ
			,P.CHGDATE 		= C.CHGDATE
			,P.CHGUSERID 	= C.CHGUSERID
			,P.CHGUSEQ 		= C.CHGUSEQ
	WHEN NOT MATCHED THEN
		INSERT (
			 P.REGSEQ
			,P.USERID
			,P.USEQ
			,P.SITE

			,P.GBN

			,P.AMT
			,P.STATUS
			,P.PRICE
			,P.ETC
			,P.SEQ
			,P.CCNT
			,P.PCNT
			,P.TCNT
			,P.PAYMODE
			,P.PAYNAME
			,P.DEPOYN
			,P.ORDERNO
			,P.REGDATE
			,P.REGUSERID
			,P.REGUSEQ
			,P.CHGDATE
			,P.CHGUSERID
			,P.CHGUSEQ
		) VALUES (
			 C.REGSEQ
			,C.USERID
			,C.USEQ
			,C.SITE
			,C.MENUCODE
			,C.CNT
			,C.GBN
			,C.USEYN
			,C.AMT
			,C.STATUS
			,C.PRICE
			,C.ETC
			,C.SEQ
			,C.CCNT
			,C.PCNT
			,C.TCNT
			,C.PAYMODE
			,C.PAYNAME
			,C.DEPOYN
			,C.ORDERNO
			,C.REGDATE
			,C.REGUSERID
			,C.REGUSEQ
			,C.CHGDATE
			,C.CHGUSERID
			,C.CHGUSEQ

	/**** 집계 테이블 사용 쿼리 위치  *****/
	/**** USP_M_CONT_USE_STATUS.M_CONT_VIEW 집계 테이블 생성시 사용  *****/
	/**** USP_M_CONT_USE_STATUS.M_CONT_VIEW_COMB 집계 테이블 생성시 사용  *****/
	/**** USP_M_CONT_USE_STATUS.M_CONT_VIEW_YEAR 집계 테이블 생성시 사용  *****/

    USP_EXCEPTION('', 'M_CONT_GGMENU', '일일배치', 'USP_M_CONT_USE_STATUS-5', 'DBMS_BATCH_SYSTEM');

				WHERE L.SITE = TC.SITE

					AND L.REGTIME = TRUNC(TC.REGTIME)
				) AS LOGIN_CNT

			,COUNT(DISTINCT TC.CSEQ) AS CSEQ_CNT

						ELSE NULL END)) AS ONLINE_CNT

						ELSE 0 END)) AS MOBILE_TIME

						ELSE 0 END)) AS TV_TIME

										AND TA.SITE = TC.SITE
										--AND NVL(TA.CORP,0) = NVL(TC.CORP,0)

			--AND TB.STATUS IN ('Y','S')

			,0 AS INSITE_ON_CNT --인사이트포럼(온라인)

			-- AND SD.ISRT_DATE >= TO_CHAR(TRUNC(SYSDATE - 2),'YYYY-MM-DD')
			AND SD.ISRT_DATE >= TRUNC(SYSDATE - 2)
			AND SD.ISRT_DATE < TRUNC(SYSDATE)
		GROUP BY

			,SD.USER_NO
			,TRUNC(SD.ISRT_DATE)

	GROUP BY

		,CV.CORP
		,CV.VIEWDATE

			,(CASE WHEN TO_CHAR(TO_DATE(CV.VIEWDATE),'YYYY') 	= (TO_CHAR(SYSDATE,'YYYY')) 		AND (CV.ONLINE_CNT + CV.MOBILE_CNT + CV.TV_CNT) > 0 THEN U.USEQ ELSE NULL END) 	AS PCTT
			,(CASE WHEN TO_CHAR(TO_DATE(CV.VIEWDATE),'YYYYMM') 	= (TO_CHAR(SYSDATE,'YYYY') || '01') AND (CV.ONLINE_CNT + CV.MOBILE_CNT + CV.TV_CNT) > 0 THEN U.USEQ ELSE NULL END) 	AS PC01
			,(CASE WHEN TO_CHAR(TO_DATE(CV.VIEWDATE),'YYYYMM') 	= (TO_CHAR(SYSDATE,'YYYY') || '02') AND (CV.ONLINE_CNT + CV.MOBILE_CNT + CV.TV_CNT) > 0 THEN U.USEQ ELSE NULL END) 	AS PC02
			,(CASE WHEN TO_CHAR(TO_DATE(CV.VIEWDATE),'YYYYMM') 	= (TO_CHAR(SYSDATE,'YYYY') || '03') AND (CV.ONLINE_CNT + CV.MOBILE_CNT + CV.TV_CNT) > 0 THEN U.USEQ ELSE NULL END) 	AS PC03
			,(CASE WHEN TO_CHAR(TO_DATE(CV.VIEWDATE),'YYYYMM') 	= (TO_CHAR(SYSDATE,'YYYY') || '04') AND (CV.ONLINE_CNT + CV.MOBILE_CNT + CV.TV_CNT) > 0 THEN U.USEQ ELSE NULL END) 	AS PC04
			,(CASE WHEN TO_CHAR(TO_DATE(CV.VIEWDATE),'YYYYMM') 	= (TO_CHAR(SYSDATE,'YYYY') || '05') AND (CV.ONLINE_CNT + CV.MOBILE_CNT + CV.TV_CNT) > 0 THEN U.USEQ ELSE NULL END) 	AS PC05
			,(CASE WHEN TO_CHAR(TO_DATE(CV.VIEWDATE),'YYYYMM') 	= (TO_CHAR(SYSDATE,'YYYY') || '06') AND (CV.ONLINE_CNT + CV.MOBILE_CNT + CV.TV_CNT) > 0 THEN U.USEQ ELSE NULL END) 	AS PC06
			,(CASE WHEN TO_CHAR(TO_DATE(CV.VIEWDATE),'YYYYMM') 	= (TO_CHAR(SYSDATE,'YYYY') || '07') AND (CV.ONLINE_CNT + CV.MOBILE_CNT + CV.TV_CNT) > 0 THEN U.USEQ ELSE NULL END) 	AS PC07
			,(CASE WHEN TO_CHAR(TO_DATE(CV.VIEWDATE),'YYYYMM') 	= (TO_CHAR(SYSDATE,'YYYY') || '08') AND (CV.ONLINE_CNT + CV.MOBILE_CNT + CV.TV_CNT) > 0 THEN U.USEQ ELSE NULL END) 	AS PC08
			,(CASE WHEN TO_CHAR(TO_DATE(CV.VIEWDATE),'YYYYMM') 	= (TO_CHAR(SYSDATE,'YYYY') || '09') AND (CV.ONLINE_CNT + CV.MOBILE_CNT + CV.TV_CNT) > 0 THEN U.USEQ ELSE NULL END) 	AS PC09
			,(CASE WHEN TO_CHAR(TO_DATE(CV.VIEWDATE),'YYYYMM') 	= (TO_CHAR(SYSDATE,'YYYY') || '10') AND (CV.ONLINE_CNT + CV.MOBILE_CNT + CV.TV_CNT) > 0 THEN U.USEQ ELSE NULL END) 	AS PC10
			,(CASE WHEN TO_CHAR(TO_DATE(CV.VIEWDATE),'YYYYMM') 	= (TO_CHAR(SYSDATE,'YYYY') || '11') AND (CV.ONLINE_CNT + CV.MOBILE_CNT + CV.TV_CNT) > 0 THEN U.USEQ ELSE NULL END) 	AS PC11
			,(CASE WHEN TO_CHAR(TO_DATE(CV.VIEWDATE),'YYYYMM') 	= (TO_CHAR(SYSDATE,'YYYY') || '12') AND (CV.ONLINE_CNT + CV.MOBILE_CNT + CV.TV_CNT) > 0 THEN U.USEQ ELSE NULL END) 	AS PC12

			-- 기간별 시청이력 회원 (TOTALTIME 기준)
			,(CASE WHEN TO_CHAR(TO_DATE(CV.VIEWDATE),'YYYY') 	= (TO_CHAR(SYSDATE,'YYYY')) 		AND (CV.ONLINE_TIME + CV.MOBILE_TIME + CV.TV_TIME) > 0 THEN U.USEQ ELSE NULL END) 	AS PTTT
			,(CASE WHEN TO_CHAR(TO_DATE(CV.VIEWDATE),'YYYYMM') 	= (TO_CHAR(SYSDATE,'YYYY') || '01') AND (CV.ONLINE_TIME + CV.MOBILE_TIME + CV.TV_TIME) > 0 THEN U.USEQ ELSE NULL END) 	AS PT01
			,(CASE WHEN TO_CHAR(TO_DATE(CV.VIEWDATE),'YYYYMM') 	= (TO_CHAR(SYSDATE,'YYYY') || '02') AND (CV.ONLINE_TIME + CV.MOBILE_TIME + CV.TV_TIME) > 0 THEN U.USEQ ELSE NULL END) 	AS PT02
			,(CASE WHEN TO_CHAR(TO_DATE(CV.VIEWDATE),'YYYYMM') 	= (TO_CHAR(SYSDATE,'YYYY') || '03') AND (CV.ONLINE_TIME + CV.MOBILE_TIME + CV.TV_TIME) > 0 THEN U.USEQ ELSE NULL END) 	AS PT03
			,(CASE WHEN TO_CHAR(TO_DATE(CV.VIEWDATE),'YYYYMM') 	= (TO_CHAR(SYSDATE,'YYYY') || '04') AND (CV.ONLINE_TIME + CV.MOBILE_TIME + CV.TV_TIME) > 0 THEN U.USEQ ELSE NULL END) 	AS PT04
			,(CASE WHEN TO_CHAR(TO_DATE(CV.VIEWDATE),'YYYYMM') 	= (TO_CHAR(SYSDATE,'YYYY') || '05') AND (CV.ONLINE_TIME + CV.MOBILE_TIME + CV.TV_TIME) > 0 THEN U.USEQ ELSE NULL END) 	AS PT05
			,(CASE WHEN TO_CHAR(TO_DATE(CV.VIEWDATE),'YYYYMM') 	= (TO_CHAR(SYSDATE,'YYYY') || '06') AND (CV.ONLINE_TIME + CV.MOBILE_TIME + CV.TV_TIME) > 0 THEN U.USEQ ELSE NULL END) 	AS PT06
			,(CASE WHEN TO_CHAR(TO_DATE(CV.VIEWDATE),'YYYYMM') 	= (TO_CHAR(SYSDATE,'YYYY') || '07') AND (CV.ONLINE_TIME + CV.MOBILE_TIME + CV.TV_TIME) > 0 THEN U.USEQ ELSE NULL END) 	AS PT07
			,(CASE WHEN TO_CHAR(TO_DATE(CV.VIEWDATE),'YYYYMM') 	= (TO_CHAR(SYSDATE,'YYYY') || '08') AND (CV.ONLINE_TIME + CV.MOBILE_TIME + CV.TV_TIME) > 0 THEN U.USEQ ELSE NULL END) 	AS PT08
			,(CASE WHEN TO_CHAR(TO_DATE(CV.VIEWDATE),'YYYYMM') 	= (TO_CHAR(SYSDATE,'YYYY') || '09') AND (CV.ONLINE_TIME + CV.MOBILE_TIME + CV.TV_TIME) > 0 THEN U.USEQ ELSE NULL END) 	AS PT09
			,(CASE WHEN TO_CHAR(TO_DATE(CV.VIEWDATE),'YYYYMM') 	= (TO_CHAR(SYSDATE,'YYYY') || '10') AND (CV.ONLINE_TIME + CV.MOBILE_TIME + CV.TV_TIME) > 0 THEN U.USEQ ELSE NULL END) 	AS PT10
			,(CASE WHEN TO_CHAR(TO_DATE(CV.VIEWDATE),'YYYYMM') 	= (TO_CHAR(SYSDATE,'YYYY') || '11') AND (CV.ONLINE_TIME + CV.MOBILE_TIME + CV.TV_TIME) > 0 THEN U.USEQ ELSE NULL END) 	AS PT11
			,(CASE WHEN TO_CHAR(TO_DATE(CV.VIEWDATE),'YYYYMM') 	= (TO_CHAR(SYSDATE,'YYYY') || '12') AND (CV.ONLINE_TIME + CV.MOBILE_TIME + CV.TV_TIME) > 0 THEN U.USEQ ELSE NULL END) 	AS PT12
		FROM US_USER U
			LEFT OUTER JOIN M_CONT_VIEW CV 	ON CV.SITE = U.SITE
											AND CV.USEQ = U.USEQ
											AND TO_CHAR(TO_DATE(CV.VIEWDATE),'YYYY') = TO_CHAR(SYSDATE,'YYYY')
											--AND ((U.SITE = 'MS' AND CV.CORP = U.CORP) OR 1 = 1)
		WHERE 1 = 1

			AND U.SITE <> 'MS'
			AND U.SITE IN (SELECT S.SITE FROM A_SITE S)

			AND U.USEYN = 'Y'
			AND SUBSTR(U.USERGBN,0,1) = 'A'
			--AND SUBSTR(U.USERGBN,0,1) IN ('A','C')

			--AND U.SIGNSEQ IS NOT NULL
			--AND NVL(U.SERVSTIME,NVL(U.OUTTIME,U.SERVETIME)) IS NOT NULL
			AND (
				TO_CHAR(SYSDATE,'YYYY') BETWEEN TO_CHAR(U.SERVSTIME,'YYYY') AND TO_CHAR(NVL(U.OUTTIME,U.SERVETIME),'YYYY')
				OR
				TO_CHAR(SYSDATE,'YYYY') = TO_CHAR(TO_DATE(CV.VIEWDATE),'YYYY')
				)

	) X
	GROUP BY
		 X.SITE
		,X.USEQ
		,X.CORP
		,X.CONT_YEAR
	;

						) B ON B.SITE = CV.SITE AND B.USEQ = CV.USEQ --AND B.CORP = CV.CORP
							AND B.VIEWYEAR = TO_CHAR(TO_DATE(CV.VIEWDATE),'YYYY')
							--AND ((CV.SITE = 'MS' AND B.CORP = CV.CORP) OR 1 = 1)
	WHERE 1 = 1
		AND CV.USEQ IS NOT NULL

		AND CV.SITE <> 'MS'
		AND CV.SITE IN (SELECT S.SITE FROM A_SITE S)

		AND TO_CHAR(TO_DATE(CV.VIEWDATE),'YYYY') = TO_CHAR(SYSDATE,'YYYY')
	GROUP BY
		 CV.SITE
		,CV.USEQ
		,CV.CORP
		,TO_CHAR(TO_DATE(CV.VIEWDATE),'YYYY')
	;

    USP_EXCEPTION('', 'USP_M_CONT_USE_STATUS 완료', '일일배치', 'USP_M_CONT_USE_STATUS-13', 'DBMS_BATCH_SYSTEM');


	COMMIT;

	EXCEPTION
			WHEN OTHERS THEN ROLLBACK;
	USP_EXCEPTION('', DBMS_UTILITY.FORMAT_ERROR_STACK, '일일배치', 'USP_M_CONT_USE_STATUS', 'DBMS_BATCH_SYSTEM');

END;
